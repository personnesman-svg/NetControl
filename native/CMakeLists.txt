cmake_minimum_required(VERSION 3.22.1)
project(netcontrol LANGUAGES CXX)

add_library(netcontrol_core STATIC src/netcore.cpp)
target_include_directories(netcontrol_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(netcontrol_core PUBLIC cxx_std_17)
target_compile_options(netcontrol_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -fstack-protector-strong)

if(ANDROID)
    add_library(netcontrol SHARED src/jni_bridge.cpp)
    target_link_libraries(netcontrol PRIVATE netcontrol_core log)
    target_compile_options(netcontrol PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
else()
    enable_testing()
    add_executable(netcore_tests tests/netcore_test.cpp)
    target_link_libraries(netcore_tests PRIVATE netcontrol_core)
    target_compile_options(netcore_tests PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
    add_test(NAME netcore_tests COMMAND netcore_tests)
endif()
